	GPIO_Struct.Mode = LL_GPIO_MODE_ALTERNATE;
	GPIO_Struct.Alternate = LL_GPIO_AF_5;
	GPIO_Struct.Speed = LL_GPIO_SPEED_FREQ_HIGH;
	GPIO_Struct.Pull = LL_GPIO_PULL_NO;

static void ADC_Setup(void){
	LL_AHB1_GRP1_EnableClock(LL_AHB1_GRP1_PERIPH_GPIOA); 
	LL_GPIO_InitTypeDef GPIO_Struct_Poti; 
	LL_GPIO_StructInit(&GPIO_Struct_Poti); 
	
	GPIO_Struct_Poti.Mode = LL_GPIO_MODE_ANALOG; 
	GPIO_Struct_Poti.OutputType = LL_GPIO_OUTPUT_PUSHPULL; 
	GPIO_Struct_Poti.Pin = LL_GPIO_PIN_7;
	GPIO_Struct_Poti.Pull = LL_GPIO_PULL_NO; 
	GPIO_Struct_Poti.Speed = LL_GPIO_SPEED_FREQ_HIGH; 

	LL_GPIO_Init(GPIOA, &GPIO_Struct_Poti); 
	
	LL_ADC_CommonInitTypeDef ADC_Common_Struct; 
	LL_ADC_REG_InitTypeDef ADC_Reg_Init_Struct; 

	LL_ADC_CommonStructInit(&ADC_Common_Struct); 
	LL_AHB1_GRP1_EnableClock(LL_AHB1_GRP1_PERIPH_ADC12); 
	LL_ADC_SetCommonClock(__LL_ADC_COMMON_INSTANCE(ADC2), LL_ADC_CLOCK_SYNC_PCLK_DIV1); 
	
	LL_ADC_REG_StructInit(&ADC_Reg_Init_Struct); 
	ADC_Reg_Init_Struct.ContinuousMode = LL_ADC_REG_CONV_CONTINUOUS;
	ADC_Reg_Init_Struct.Overrun = LL_ADC_REG_OVR_DATA_OVERWRITTEN; 
	ADC_Reg_Init_Struct.SequencerLength = LL_ADC_REG_SEQ_SCAN_DISABLE; 
	LL_ADC_REG_Init(ADC2, &ADC_Reg_Init_Struct); 
	
	
	LL_ADC_SetResolution(ADC2, LL_ADC_RESOLUTION_12B);
	LL_ADC_SetDataAlignment(ADC2, LL_ADC_DATA_ALIGN_RIGHT); 
	LL_ADC_REG_SetSequencerRanks(ADC2, LL_ADC_REG_RANK_1, LL_ADC_CHANNEL_4);
	LL_ADC_SetChannelSamplingTime(ADC2, LL_ADC_CHANNEL_4, LL_ADC_SAMPLINGTIME_61CYCLES_5); 
	
	if(LL_ADC_IsEnabled(ADC2) == 0){
		LL_ADC_EnableInternalRegulator(ADC2); 
		
		LL_TIM_EnableCounter (TIM16); 
		if(LL_TIM_IsActiveFlag_UPDATE(TIM16)){
			LL_TIM_ClearFlag_UPDATE(TIM16); 	
		}
		
		while(!LL_TIM_IsActiveFlag_UPDATE(TIM16)){}; 
			LL_TIM_DisableCounter(TIM16);
			LL_TIM_ClearFlag_UPDATE(TIM16); 
			NVIC_ClearPendingIRQ(TIM16_IRQn); 
			
			LL_ADC_StartCalibration(ADC2, LL_ADC_SINGLE_ENDED); 
			
			while(LL_ADC_IsCalibrationOnGoing(ADC2) != 0){}; 
			
			uint32_t calibration_value = LL_ADC_GetCalibrationFactor(ADC2, LL_ADC_SINGLE_ENDED); 

	}
	
	LL_ADC_Enable(ADC2); 
				
	while(!LL_ADC_IsActiveFlag_ADRDY(ADC2)){}; 
	LL_ADC_REG_StartConversion(ADC2); 

}

  LL_TIM_InitTypeDef TIM_Struct;

	TIM_Struct.Autoreload = 0xFFFF;
	LL_TIM_Init(TIM3, &TIM_Struct);

	LL_TIM_EnableCounter(TIM3);
	

}
	
void EXTI0_IRQHandler(void){
	if(LL_EXTI_IsActiveFlag_0_31(LL_EXTI_LINE_0)){
		LL_EXTI_ClearFlag_0_31(LL_EXTI_LINE_0);
		
		uint16_t period_ticks = LL_TIM_GetCounter(TIM3);
		
		LL_TIM_SetCounter(TIM3, 0);
		
		motor_period_us = period_ticks;
		
		if (motor_period_us > 0){
			motor_rpm = 60000000 / motor_period_us;
		}
		
		t = 0;
	}
}

void USART2_IRQHandler(void){
	if(LL_USART_IsActiveFlag_RXNE(USART2)){
		uint8_t recvd = LL_USART_ReceiveData8(USART2);
		if((recvd & 0xF0) == 0x70 | (recvd & 0xF0) == 0x90){
			SPI_Transmit_Speed(recvd);
		}
		else if((recvd & 0xC0) == 0xC0){
			if(uart_led_index < 3){
				uart_led_buffer[uart_led_index] = recvd;
				uart_led_index++;
			}
		}
		else if((recvd & 0xF8) == 0x10){
			if(uart_led_index == 3){
				SPI_Transmit_LED(uart_led_buffer[0], uart_led_buffer[1], uart_led_buffer[2], recvd);
			}
			uart_led_index = 0;
		}
	}
	
	if(LL_USART_IsActiveFlag_TXE(USART2)){
		LL_USART_TransmitData8(USART2, uart_tx_buffer[uart_tx_index]);
		uart_tx_index++;
		if (uart_tx_index >= uart_tx_length){
			LL_USART_DisableIT_TXE(USART2);
			uart_is_transmitting = 0;
		}
	}
}

void SPI1_IRQHandler(void){
	if (LL_SPI_IsActiveFlag_RXNE(SPI1)){
		volatile uint8_t dummy = LL_SPI_ReceiveData8(SPI1);
		spi_bytes_sent++; 
		if( spi_bytes_sent < spi_target_bytes) {
				LL_SPI_TransmitData8(SPI1, data_bytes[spi_bytes_sent]);
		}else{
			Trigger_Strobe();
			
			LL_SPI_DisableIT_RXNE(SPI1);
			
			spi_busy = 0;
		}
	}
}

void TIM2_IRQHandler(void) {
	if(LL_TIM_IsActiveFlag_UPDATE(TIM2) == 1){
		LL_TIM_ClearFlag_UPDATE(TIM2);
		
		int adc_value = 0; 
		float voltage = 0; 
		
		adc_value = LL_ADC_REG_ReadConversionData12(ADC2); 
		voltage = ((float)adc_value)/(4095.0f) * 3.3f; 
		
		tim2_uart_divider++;
		
		if (tim2_uart_divider >= 10000){
			tim2_uart_divider = 0;
			if(!uart_is_transmitting){
				uart_is_transmitting = 1;
				sprintf(uart_tx_buffer, "RPM: %u\r\n", motor_rpm);
				uart_tx_length = strlen(uart_tx_buffer);
				uart_tx_index = 0;
				
				LL_USART_EnableIT_TXE(USART2);
			}
		}
		
			
		
		uint8_t d1 = 0xC0;
		uint8_t d2 = 0xC0;
		uint8_t d3 = 0xC0;
		uint8_t cmd = 0;
		
		uint8_t module = running_pos / 12;
		uint8_t col = running_pos % 12;
		uint8_t is_odd = running_pos % 2;
		uint8_t bit_index = col/2;
		
		uint8_t active_data = 0xC0 | (1 << bit_index);
		
		if(module == 0) d3 = active_data;
		else if (module == 1) d2 = active_data;
		else if (module == 2) d1 = active_data;
			
		if (is_odd == 0){ // ROWS PROJECTED UPSIDE DOWN (RED EVEN ON THE BOTTOM; BLUE ODD ON THE TOP)
			cmd = 0x14;
		} else {
			cmd = 0x15;
		}
		
		if(!spi_busy){
			SPI_Transmit_LED(d1, d2, d3, cmd); // LEFT MOST LED MODULE IS RIGHT MOST DATA COMMAND
		}
		
		uint32_t delay_threshold = (4095 - adc_value)/100 + 20;
			
		animation_delay_counter++;
			
		if(animation_delay_counter >= delay_threshold) {
			animation_delay_counter = 0;
			
			running_pos++;
			if(running_pos >= 36){
				running_pos = 0;
			}
		}
		
	}
}



void SPI_Transmit_Speed(uint8_t data){
	if (spi_busy) return;
	spi_busy = 1;
	spi_bytes_sent = 0;
	spi_target_bytes = 1;
	data_bytes[0] = data; 
	LL_SPI_EnableIT_RXNE(SPI1);
	
	LL_SPI_TransmitData8(SPI1, data_bytes[0]);

}


void SPI_Transmit_LED(uint8_t d1, uint8_t d2, uint8_t d3, uint8_t d4){
	if (spi_busy) return;
	spi_busy = 1;
	spi_bytes_sent = 0; 
	spi_target_bytes = 4; 
	data_bytes[0] = d1; 
	data_bytes[1] = d2; 
	data_bytes[2] = d3; 
	data_bytes[3] = d4; 
	LL_SPI_EnableIT_RXNE(SPI1);
	
	LL_SPI_TransmitData8(SPI1, data_bytes[0]);

}

	//for(int i = 0; i < 1000; i++){__NOP();} 
	LL_GPIO_ResetOutputPin(GPIOB, LL_GPIO_PIN_1);
}

/**
  * @brief  System Clock Configuration
  *         The system Clock is configured as follow : 
  *            System Clock source            = PLL (HSI)
  *            SYSCLK(Hz)                     = 64000000
  *            HCLK(Hz)                       = 64000000
  *            AHB Prescaler                  = 1
  *            APB1 Prescaler                 = 2
  *            APB2 Prescaler                 = 1
  *            PLLMUL                         = 16
  *            Flash Latency(WS)              = 2
  * @param  None
  * @retval None
  */
void SystemClock_Config(void)
{
  /* Set FLASH latency */ 
  LL_FLASH_SetLatency(LL_FLASH_LATENCY_2);

  /* Enable HSI if not already activated*/
  if (LL_RCC_HSI_IsReady() == 0)
  {
    /* Enable HSI and wait for activation*/
    LL_RCC_HSI_Enable(); 
    while(LL_RCC_HSI_IsReady() != 1)
    {
    };
  }

  /* Main PLL configuration and activation */
  LL_RCC_PLL_ConfigDomain_SYS(LL_RCC_PLLSOURCE_HSI_DIV_2, LL_RCC_PLL_MUL_16);
  
  LL_RCC_PLL_Enable();
  while(LL_RCC_PLL_IsReady() != 1) 
  {
  };
  
  /* Sysclk activation on the main PLL */
  LL_RCC_SetAHBPrescaler(LL_RCC_SYSCLK_DIV_1);
  LL_RCC_SetSysClkSource(LL_RCC_SYS_CLKSOURCE_PLL);
  while(LL_RCC_GetSysClkSource() != LL_RCC_SYS_CLKSOURCE_STATUS_PLL)
  {
  };
  
  /* Set APB1 & APB2 prescaler*/
  LL_RCC_SetAPB1Prescaler(LL_RCC_APB1_DIV_2);
  LL_RCC_SetAPB2Prescaler(LL_RCC_APB2_DIV_1);
  
  /* Set systick to 1ms in using frequency set to 64MHz */
  /* This frequency can be calculated through LL RCC macro */
  /* ex: __LL_RCC_CALC_PLLCLK_FREQ ((HSI_VALUE / 2), LL_RCC_PLL_MUL_16) */
  LL_Init1msTick(64000000);
  
  /* Update CMSIS variable (which can be updated also through SystemCoreClockUpdate function) */
  LL_SetSystemCoreClock(64000000);
}
